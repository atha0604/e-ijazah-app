# 💻 PERINTAH CMD MANUAL
# Untuk Copy-Paste ke Command Prompt

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

## CARA 1: ONE-LINER SUPER CEPAT (Recommended)

Buka CMD (Windows + R → ketik: cmd → Enter), lalu copy-paste:

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

### Versi Pendek (Download + Extract + Install):

powershell -Command "cd C:\; Invoke-WebRequest -Uri 'https://github.com/atha0604/e-ijazah-app/archive/refs/heads/brave-black.zip' -OutFile 'e-ijazah.zip'; Expand-Archive 'e-ijazah.zip' 'C:\E-Ijazah' -Force; cd 'C:\E-Ijazah\e-ijazah-app-brave-black'; npm install; echo 'Instalasi selesai! Jalankan: npm start'"

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

## CARA 2: GIT CLONE (Untuk yang sudah install Git)

cd C:\
git clone -b brave-black https://github.com/atha0604/e-ijazah-app
cd e-ijazah-app
npm install
npm start

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

## CARA 3: STEP-BY-STEP MANUAL

### Step 1: Buat Folder
mkdir C:\E-Ijazah
cd C:\E-Ijazah

### Step 2: Download (PowerShell)
powershell -Command "Invoke-WebRequest -Uri 'https://github.com/atha0604/e-ijazah-app/archive/refs/heads/brave-black.zip' -OutFile 'e-ijazah.zip'"

### Step 3: Extract
powershell -Command "Expand-Archive -Path 'e-ijazah.zip' -DestinationPath '.' -Force"

### Step 4: Masuk ke Folder
cd e-ijazah-app-brave-black

### Step 5: Install Dependencies
npm install

### Step 6: Jalankan
npm start

### Step 7: Buka Browser
start http://localhost:3000

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

## PERINTAH TROUBLESHOOTING

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

### Cek Node.js & NPM:
node --version
npm --version

### Cek Git:
git --version

### Cek Port 3000:
netstat -ano | findstr :3000

### Kill Process di Port 3000:
npx kill-port 3000

### Atau manual:
tasklist | findstr node
taskkill /F /PID [PID_NUMBER]

### Clear NPM Cache:
npm cache clean --force

### Reinstall Dependencies:
rmdir /s /q node_modules
del package-lock.json
npm install

### Cek Koneksi Internet:
ping github.com

## PERINTAH BACKUP & RESTORE

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

### Backup Database:
cd C:\E-Ijazah\e-ijazah-app-brave-black
mkdir backup\manual-%DATE:/=-%
copy src\database\db.sqlite backup\manual-%DATE:/=-%\db.sqlite.backup

### Restore Database:
copy backup\[tanggal]\db.sqlite.backup src\database\db.sqlite

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
